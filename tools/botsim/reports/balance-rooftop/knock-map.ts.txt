import { createBots } from '@line-war/bots';
import { levelById } from '@line-war/content';
import { createMatch, makeConfig, NO_INPUT, step, type MatchInputs } from '@line-war/sim';
const level = levelById(process.argv[2] ?? 'rooftop-garden');
const N = Number(process.argv[3] ?? 30);
const grid = new Map<string, number>();
const items = new Map<string, number>();
const inv = new Map<string, number>(); // victim side relation: near which crate
let knocks = 0, rounds = 0, carryTicks = 0, marketCarryTicks = 0;
const W = level.arena.rows[0]!.length;
for (let s = 1; s <= N; s++) {
  const cfg = makeConfig({ seed: 1000 + s });
  const st = createMatch({ arena: level.arena, recipes: level.recipes, ingredients: level.ingredients, charms: level.charms, charmsPerRound: level.charmsPerRound, config: cfg });
  const brains = createBots([0, 1, 2, 3].map((i) => ({ difficulty: 'normal' as const, seed: s * 10 + i })));
  let lastPhase = '';
  while (st.round.phase !== 'matchOver') {
    const inputs = [0, 1, 2, 3].map((i) => brains[i]?.decide(st, i as 0 | 1 | 2 | 3) ?? NO_INPUT) as MatchInputs;
    step(st, inputs);
    if (st.round.phase !== lastPhase) { if (st.round.phase === 'service') rounds++; lastPhase = st.round.phase; }
    for (const p of st.players) if (p.holding && ['service','rush','lastCall'].includes(st.round.phase)) { carryTicks++; const ti = Math.floor(p.pos.y) * W + Math.floor(p.pos.x); if (st.arena.zones[ti] === 'market') marketCarryTicks++; }
    for (const ev of st.events) if (ev.type === 'knock') {
      knocks++;
      const v = st.players[ev.victim]!; const a = st.players[ev.by]!;
      const k = `${Math.floor(v.pos.x)},${Math.floor(v.pos.y)}`; grid.set(k, (grid.get(k) ?? 0) + 1);
      const fi = st.floorItems.find((f) => f.item.id === ev.itemId);
      const it = fi?.item; const name = it ? (it.type === 'ingredient' ? it.kind : it.type === 'plate' ? 'plate:' + it.contents.map((c: any) => c.kind).join('+') : it.type) : '?';
      items.set(name, (items.get(name) ?? 0) + 1);
      // victim heading home or away? victim team side
      const home = v.team === 0 !== st.round.swapped ? 'L' : 'R';
      const dir = (v.vel.x < 0 ? 'L' : v.vel.x > 0 ? 'R' : '0');
      const key = `victimHome=${home} moving=${dir} attackerHolding=${a.holding ? 'y' : 'n'}`;
      inv.set(key, (inv.get(key) ?? 0) + 1);
    }
  }
}
console.log('rounds', rounds, 'knocks/round', (knocks / rounds).toFixed(2), 'marketCarrySec/round', (marketCarryTicks / 30 / rounds).toFixed(1), 'knocks per market-carry-minute', (knocks / (marketCarryTicks / 30 / 60)).toFixed(2));
console.log('items', [...items].sort((a, b) => b[1] - a[1]));
console.log('rel', [...inv].sort((a, b) => b[1] - a[1]).slice(0, 12));
const H = level.arena.rows.length;
for (let y = 0; y < H; y++) { let line = ''; for (let x = 0; x < W; x++) { const n = grid.get(`${x},${y}`) ?? 0; line += n ? String(Math.min(999, Math.round(n / rounds * 100))).padStart(4) : '   ' + level.arena.rows[y]![x]; } console.log(line); }
console.log('(cells = knocks per 100 rounds)');
